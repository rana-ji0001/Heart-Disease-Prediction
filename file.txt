    <script>
      const chartLabels = {{ chart_labels | default([]) | tojson | safe }};
      const chartValues = {{ chart_values | default([]) | tojson | safe }};
      const accLabels   = {{ acc_labels   | default([]) | tojson | safe }};
      const accValues   = {{ acc_values   | default([]) | tojson | safe }};
      const percent     = parseFloat("{{ percent }}") || 0;

      /* ── palette ── */
      const PURPLES = ['rgba(102,126,234,0.85)','rgba(118,75,162,0.85)','rgba(86,204,242,0.85)'];
      const ORANGES = ['rgba(230,126,34,0.85)','rgba(46,204,113,0.85)','rgba(231,76,60,0.85)'];
      const BORDERS_P = PURPLES.map(c => c.replace('0.85','1'));
      const BORDERS_O = ORANGES.map(c => c.replace('0.85','1'));

      const baseScales = {
        y: {
          beginAtZero: true, max: 100,
          grid: { color: 'rgba(255,255,255,0.05)' },
          ticks: { color: '#b0b0c0', font: { family: 'Poppins' } }
        },
        x: {
          grid: { color: 'rgba(255,255,255,0.05)' },
          ticks: { color: '#b0b0c0', font: { family: 'Poppins' } }
        }
      };

      const baseLegend = {
        labels: { color: '#e0e0e0', font: { family: 'Poppins', size: 12 } }
      };

      /* ── Gauge ── */
      (function buildGauge() {
        const p = Math.min(Math.max(percent, 0), 100);
        const canvas = document.getElementById('gaugeChart');
        const ctx = canvas.getContext('2d');
        const W = 320, H = 180;
        canvas.width = W; canvas.height = H;

        const cx = W / 2, cy = H - 20, r = 120;
        const startA = Math.PI, endA = 2 * Math.PI;
        const fillA = startA + (p / 100) * Math.PI;

        // bg arc
        ctx.beginPath();
        ctx.arc(cx, cy, r, startA, endA);
        ctx.lineWidth = 22;
        ctx.strokeStyle = 'rgba(255,255,255,0.07)';
        ctx.lineCap = 'round';
        ctx.stroke();

        // colored fill
        const grad = ctx.createLinearGradient(cx - r, cy, cx + r, cy);
        grad.addColorStop(0, '#2ecc71');
        grad.addColorStop(0.5, '#f1c40f');
        grad.addColorStop(1, '#e74c3c');
        ctx.beginPath();
        ctx.arc(cx, cy, r, startA, fillA);
        ctx.lineWidth = 22;
        ctx.strokeStyle = grad;
        ctx.lineCap = 'round';
        ctx.stroke();

        // needle
        const angle = startA + (p / 100) * Math.PI;
        const nx = cx + (r - 14) * Math.cos(angle);
        const ny = cy + (r - 14) * Math.sin(angle);
        ctx.beginPath();
        ctx.moveTo(cx, cy);
        ctx.lineTo(nx, ny);
        ctx.lineWidth = 3;
        ctx.strokeStyle = '#fff';
        ctx.lineCap = 'round';
        ctx.stroke();

        // center dot
        ctx.beginPath();
        ctx.arc(cx, cy, 6, 0, 2 * Math.PI);
        ctx.fillStyle = '#fff';
        ctx.fill();

        // percent text
        ctx.font = '800 40px Poppins';
        ctx.fillStyle = '#e0e0e0';
        ctx.textAlign = 'center';
        ctx.fillText(p.toFixed(1) + '%', cx, cy - 18);

        // tick labels
        ['0%','25%','50%','75%','100%'].forEach((label, i) => {
          const a = Math.PI + (i / 4) * Math.PI;
          const tx = cx + (r + 18) * Math.cos(a);
          const ty = cy + (r + 18) * Math.sin(a);
          ctx.font = '11px Poppins';
          ctx.fillStyle = '#6a6a8a';
          ctx.textAlign = 'center';
          ctx.fillText(label, tx, ty);
        });

        // badge
        const badge = document.getElementById('riskBadge');
        if (p < 35) {
          badge.textContent = 'LOW RISK'; badge.className = 'risk-badge risk-low';
        } else if (p < 65) {
          badge.textContent = 'MODERATE RISK'; badge.className = 'risk-badge risk-medium';
        } else {
          badge.textContent = 'HIGH RISK'; badge.className = 'risk-badge risk-high';
        }
      })();

      /* ── Stat cards ── */
      (function buildStats() {
        const cards = document.getElementById('statCards');
        const avgAcc = accValues.length
          ? (accValues.reduce((a,b) => a + b, 0) / accValues.length).toFixed(1)
          : null;

        const allVals = [...chartValues, ...accValues];
        if (allVals.length === 0) return;

        document.getElementById('sc-risk').textContent    = percent.toFixed(1);
        document.getElementById('sc-models').textContent  = Math.max(chartLabels.length, accLabels.length) || '—';
        document.getElementById('sc-acc').textContent     = avgAcc ? avgAcc + '%' : '—';

        if (chartValues.length > 1) {
          const max = Math.max(...chartValues), min = Math.min(...chartValues);
          const spread = (max - min).toFixed(1);
          document.getElementById('sc-consensus').textContent = spread < 10 ? 'High' : spread < 25 ? 'Medium' : 'Low';
        } else {
          document.getElementById('sc-consensus').textContent = '—';
        }
        cards.style.display = 'grid';
      })();

      /* ── Bar chart: model probabilities ── */
      const modelCtx = document.getElementById('modelChart');
      if (modelCtx && chartLabels.length && chartValues.length) {
        new Chart(modelCtx, {
          type: 'bar',
          data: {
            labels: chartLabels,
            datasets: [{
              label: 'Predicted Probability (%)',
              data: chartValues,
              backgroundColor: PURPLES,
              borderColor: BORDERS_P,
              borderWidth: 2,
              borderRadius: 8,
              hoverBackgroundColor: BORDERS_P
            }]
          },
          options: {
            responsive: true, maintainAspectRatio: false,
            scales: baseScales,
            plugins: {
              legend: baseLegend,
              tooltip: {
                callbacks: { label: ctx => ' ' + ctx.parsed.y.toFixed(1) + '%' }
              }
            },
            animation: { duration: 1400, easing: 'easeInOutQuart' }
          }
        });
      }

      /* ── Doughnut chart ── */
      const doughCtx = document.getElementById('doughnutChart');
      if (doughCtx && chartLabels.length && chartValues.length) {
        new Chart(doughCtx, {
          type: 'doughnut',
          data: {
            labels: chartLabels,
            datasets: [{
              data: chartValues,
              backgroundColor: PURPLES,
              borderColor: BORDERS_P,
              borderWidth: 2,
              hoverOffset: 8
            }]
          },
          options: {
            responsive: true, maintainAspectRatio: false,
            cutout: '65%',
            plugins: {
              legend: { ...baseLegend, position: 'bottom' },
              tooltip: {
                callbacks: { label: ctx => ' ' + ctx.label + ': ' + ctx.parsed.toFixed(1) + '%' }
              }
            },
            animation: { duration: 1400, easing: 'easeInOutQuart' }
          }
        });
      }

      /* ── Bar chart: model accuracies ── */
      const accCtx = document.getElementById('accuracyChart');
      if (accCtx && accLabels.length && accValues.length) {
        new Chart(accCtx, {
          type: 'bar',
          data: {
            labels: accLabels,
            datasets: [{
              label: 'Model Accuracy (%)',
              data: accValues,
              backgroundColor: ORANGES,
              borderColor: BORDERS_O,
              borderWidth: 2,
              borderRadius: 8,
              hoverBackgroundColor: BORDERS_O
            }]
          },
          options: {
            responsive: true, maintainAspectRatio: false,
            scales: baseScales,
            plugins: {
              legend: baseLegend,
              tooltip: {
                callbacks: { label: ctx => ' ' + ctx.parsed.y.toFixed(1) + '%' }
              }
            },
            animation: { duration: 1400, easing: 'easeInOutQuart' }
          }
        });
      }

      /* ── Radar chart ── */
      const radarCtx = document.getElementById('radarChart');
      if (radarCtx && accLabels.length && accValues.length) {
        new Chart(radarCtx, {
          type: 'radar',
          data: {
            labels: accLabels,
            datasets: [{
              label: 'Accuracy (%)',
              data: accValues,
              backgroundColor: 'rgba(102,126,234,0.2)',
              borderColor: 'rgba(102,126,234,1)',
              borderWidth: 2,
              pointBackgroundColor: 'rgba(118,75,162,1)',
              pointRadius: 5
            }]
          },
          options: {
            responsive: true, maintainAspectRatio: false,
            scales: {
              r: {
                beginAtZero: true, max: 100,
                grid: { color: 'rgba(255,255,255,0.07)' },
                angleLines: { color: 'rgba(255,255,255,0.07)' },
                pointLabels: { color: '#b0b0c0', font: { family: 'Poppins', size: 11 } },
                ticks: { color: '#b0b0c0', backdropColor: 'transparent', stepSize: 25 }
              }
            },
            plugins: {
              legend: baseLegend,
              tooltip: {
                callbacks: { label: ctx => ' ' + ctx.parsed.r.toFixed(1) + '%' }
              }
            },
            animation: { duration: 1400, easing: 'easeInOutQuart' }
          }
        });
      }

      /* ── Feature/accuracy progress bars ── */
      (function buildBars() {
        const wrap = document.getElementById('featureBars');
        if (!wrap || !accLabels.length) return;
        const maxVal = Math.max(...accValues);
        accLabels.forEach((label, i) => {
          const pct = accValues[i] || 0;
          const width = maxVal > 0 ? ((pct / maxVal) * 100).toFixed(1) : 0;
          wrap.innerHTML += `
            <div class="feature-bar-wrap">
              <div class="feature-bar-label"><span>${label}</span><span>${pct.toFixed(1)}%</span></div>
              <div class="feature-bar-bg">
                <div class="feature-bar-fill" style="width:${width}%"></div>
              </div>
            </div>`;
        });
      })();
    </script>
